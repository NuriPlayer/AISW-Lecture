import tensorflow as tf

myModel = tf.keras.models.Sequential([
    tf.keras.layers.Dense(64, activation='tanh'),
    tf.keras.layers.Dense(128, activation='tanh'),
    tf.keras.layers.Dense(1, activation='sigmoid'),
])

myModel.compile(optimizer='adam', 
                loss='binary_crossentropy', 
                metrics=['accuracy'])

import numpy as np
import pandas as pd

try:
    df = pd.read_csv('./data/01_Gpascore_dataset.csv')
    # print(df.head(5))
except:
    print("뭔가 잘 안되었으니 경로명 확인")
    
df = df.dropna()
print(df.isnull().sum())

x_data = df[['gre', 'gpa', 'rank']].values
y_data = df['admit'].values
# print(x_data[:5])
# print(y_data[:5])

# fit() 메서드에 넣기 전 자료형 변환해 주기
x_data = np.array(x_data, dtype=np.float32)
y_data = np.array(y_data, dtype=np.float32)

# 데이터 학습
myHistory = myModel.fit(x_data, y_data, epochs=5000, batch_size=32, 
            validation_split=0.2)

# 내 데이터 넣고 예측해 보기
# x_test = np.array([[800, 3.5, 3], [310, 3.0, 2]])
# y_test = myModel.predict(x_test)
# print(y_test)

# 데이터 시각화
import matplotlib.pyplot as plt
# 학습 손실과 검증 손실 점 그래프
plt.figure(figsize=(10, 6))
plt.scatter(range(len(myHistory.history['loss'])), myHistory.history['loss'], label='Train Loss', color='red', alpha=0.7)
plt.scatter(range(len(myHistory.history['val_loss'])), myHistory.history['val_loss'], label='Validation Loss', color='pink', alpha=0.7)
plt.title('Train vs Validation Loss')
plt.xlabel('Epochs')
plt.ylabel('Loss')
plt.legend()
plt.grid(True)

# 학습 정확도와 검증 정확도 점 그래프
plt.figure(figsize=(10, 6))
plt.scatter(range(len(myHistory.history['accuracy'])), myHistory.history['accuracy'], label='Train Accuracy', color='blue', alpha=0.7)
plt.scatter(range(len(myHistory.history['val_accuracy'])), myHistory.history['val_accuracy'], label='Validation Accuracy', color='skyblue', alpha=0.7)
plt.title('Train vs Validation Accuracy')
plt.xlabel('Epochs')
plt.ylabel('Accuracy')
plt.legend()
plt.grid(True)

# 그래프 출력
plt.show()
