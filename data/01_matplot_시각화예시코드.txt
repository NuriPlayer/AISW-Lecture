import matplotlib.pyplot as plt

# 학습 손실과 검증 손실 점 그래프
plt.figure(figsize=(10, 6))
plt.scatter(range(len(myHistory.history['loss'])), myHistory.history['loss'], label='Train Loss', color='red', alpha=0.7)
plt.scatter(range(len(myHistory.history['val_loss'])), myHistory.history['val_loss'], label='Validation Loss', color='pink', alpha=0.7)
plt.title('Train vs Validation Loss')
plt.xlabel('Epochs')
plt.ylabel('Loss')
plt.legend()
plt.grid(True)

# 학습 정확도와 검증 정확도 점 그래프
plt.figure(figsize=(10, 6))
plt.scatter(range(len(myHistory.history['accuracy'])), myHistory.history['accuracy'], label='Train Accuracy', color='blue', alpha=0.7)
plt.scatter(range(len(myHistory.history['val_accuracy'])), myHistory.history['val_accuracy'], label='Validation Accuracy', color='skyblue', alpha=0.7)
plt.title('Train vs Validation Accuracy')
plt.xlabel('Epochs')
plt.ylabel('Accuracy')
plt.legend()
plt.grid(True)

# 그래프 출력
plt.show()
